# Compile the production build orchestration and render path through progress
# persistence; only the section parser, SD, renderer and platform are substituted.
set(READER_SOURCE ${REPO_ROOT}/src/activities/reader/EpubReaderActivity.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${READER_SOURCE})
file(READ ${READER_SOURCE} source)
set(READER_HEADER ${REPO_ROOT}/src/activities/reader/EpubReaderActivity.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${READER_HEADER})
file(READ ${READER_HEADER} header)
set(constants "")
foreach(name BUILD_PAGES_PER_CHUNK BACKGROUND_BUILD_PAGES_PER_TICK BACKGROUND_BUILD_MIN_FREE_HEAP
    RENDER_MIN_FREE_HEAP BACKGROUND_BUILD_MIN_MAX_ALLOC BUILD_WINDOW_AHEAD PARTIAL_REBUILD_START_MARGIN BUILD_POPUP_PAGE_THRESHOLD
    BUILD_POPUP_BYTE_THRESHOLD BUILD_POPUP_DEADLINE_MS)
  string(REGEX MATCH "static constexpr [^;\n]+ ${name} = [^;]+;" declaration "${header}")
  if(NOT declaration)
    message(FATAL_ERROR "Incremental reader constant moved: ${name}")
  endif()
  string(APPEND constants "${declaration}\n")
endforeach()
string(REGEX MATCH "bool hasPendingSectionJump\\(\\) const \\{[^}]+\\}" pending_helper "${header}")
if(NOT pending_helper)
  message(FATAL_ERROR "Incremental reader pending helper moved")
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ReaderIncrementalConstants.h "${constants}\n${pending_helper}\n")
function(incremental_source start_marker end_marker output)
  string(FIND "${source}" "${start_marker}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Incremental reader start marker moved: ${start_marker}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 remainder)
  string(FIND "${remainder}" "${end_marker}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Incremental reader end marker moved: ${end_marker}")
  endif()
  string(SUBSTRING "${remainder}" 0 ${end} result)
  set(${output} "${result}" PARENT_SCOPE)
endfunction()
incremental_source("bool EpubReaderActivity::buildTickHeapGate()" "void EpubReaderActivity::openDictionaryWordSelect()" build_methods)
incremental_source("bool EpubReaderActivity::skipLoopDelay()" "\n  showPendingSyncSaveError();" render_path)
incremental_source("bool EpubReaderActivity::applyDeferredReposition()" "bool EpubReaderActivity::saveProgress(" deferred_methods)
incremental_source("bool EpubReaderActivity::pageTurn(" "bool EpubReaderActivity::handleLoadFailure()" page_navigation)
incremental_source("  constexpr unsigned long IDLE_PREWARM_DEBOUNCE_MS" "\n  advanceSectionBuild();" idle_prewarm)

  "#include \"ReaderIncrementalFixture.h\"\n${build_methods}\n${deferred_methods}\n${page_navigation}\n${render_path}\n}\nvoid EpubReaderActivity::runIdlePrewarm() {\n${idle_prewarm}\n}\n")
