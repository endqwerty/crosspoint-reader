function(bookmark_source file start_marker end_marker output)
  set(path ${REPO_ROOT}/src/activities/reader/${file})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${path})
  file(READ ${path} source)
  string(FIND "${source}" "${start_marker}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Bookmark action start marker moved: ${start_marker}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 remainder)
  string(FIND "${remainder}" "${end_marker}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Bookmark action end marker moved: ${end_marker}")
  endif()
  string(SUBSTRING "${remainder}" 0 ${end} body)
  set(${output} "${body}" PARENT_SCOPE)
endfunction()
