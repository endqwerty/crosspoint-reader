cmake_minimum_required(VERSION 3.16)
project(parser_failure_audit C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT GTEST_SOURCE_DIR)
  message(FATAL_ERROR "Pass -DGTEST_SOURCE_DIR=/path/to/cached/googletest-src")
endif()
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
add_subdirectory("${GTEST_SOURCE_DIR}" googletest)
add_library(crosspoint_test_common INTERFACE)
target_include_directories(crosspoint_test_common INTERFACE ${REPO_ROOT} ${REPO_ROOT}/lib)
target_compile_options(crosspoint_test_common INTERFACE -Wall -Wextra -pedantic)
enable_testing()
include(GoogleTest)
add_subdirectory(${REPO_ROOT}/test/chapter_html_slim_parser chapter_html_slim_parser)
